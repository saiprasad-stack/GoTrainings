│   │

│   └── store/             # generated by sqlc

│   └── queries.sql        # queries

### Step 1: `sql/schema.sql`

```sql
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    name TEXT NOT NULL,
    email TEXT NOT NULL UNIQUE
);
```

---

### Step 2: `sql/queries.sql`

```sql
-- name: CreateUser :exec
INSERT INTO users (name, email)
VALUES ($1, $2);

-- name: GetUser :one
SELECT id, name, email FROM users
WHERE id = $1;

-- name: GetAllUsers :many
SELECT id, name, email FROM users;

-- name: UpdateUser :exec
UPDATE users SET name = $1 WHERE id = $2;

-- name: DeleteUser :exec
DELETE FROM users WHERE id = $1;
```

---

### Step 3: `sqlc.yaml`

```yaml
version: "2"

sql:
  - schema: "sql/schema.sql"
    queries: "sql/queries.sql"
    engine: "postgresql"
    gen:
      go:
        package: "store"
        out: "internal/store"
```

---

### Step 4: Generate Code
Run
---
```
bash
sqlc generate
```

This creates:

* `internal/store/models.go`
* `internal/store/queries.sql.go`
* Typed methods like:

  * `CreateUser(ctx, params)`
  * `GetUser(ctx, id)`

### Step 5: DB Connection (`internal/db/db.go`)

```go
package db

import (
	"database/sql"
	_ "github.com/lib/pq"
)

func InitDB() (*sql.DB, error) {
	connStr := "host=localhost port=5432 user=postgres password=postgres dbname=testdb sslmode=disable"

	db, err := sql.Open("postgres", connStr)
	if err != nil {
		return nil, err
	}

	if err := db.Ping(); err != nil {
		return nil, err
	}

	return db, nil
}
```

### Step 6: Service Layer (`internal/service/user_service.go`)

```go
package service

import (
	"context"
	"log"

	"go-sqlc-app/internal/store"
)

type UserService struct {
	q *store.Queries
}

func NewUserService(q *store.Queries) *UserService {
	return &UserService{q: q}
}

func (s *UserService) CreateUser(name, email string) {
	err := s.q.CreateUser(context.Background(), store.CreateUserParams{
		Name:  name,
		Email: email,
	})
	if err != nil {
		log.Println("CreateUser error:", err)
	}
}

func (s *UserService) GetUser(id int32) store.User {
	user, err := s.q.GetUser(context.Background(), id)
	if err != nil {
		log.Println("GetUser error:", err)
	}
	return user
}

func (s *UserService) GetAllUsers() []store.User {
	users, err := s.q.GetAllUsers(context.Background())
	if err != nil {
		log.Println("GetAllUsers error:", err)
	}
	return users
}

func (s *UserService) UpdateUser(id int32, name string) {
	err := s.q.UpdateUser(context.Background(), store.UpdateUserParams{
		Name: name,
		ID:   id,
	})
	if err != nil {
		log.Println("UpdateUser error:", err)
	}
}

func (s *UserService) DeleteUser(id int32) {
	err := s.q.DeleteUser(context.Background(), id)
	if err != nil {
		log.Println("DeleteUser error:", err)
	}
}
```

### Step 7: Main (`cmd/main.go`)

```go
package main

import (
	"fmt"
	"log"

	"go-sqlc-app/internal/db"
	"go-sqlc-app/internal/service"
	"go-sqlc-app/internal/store"
)

func main() {
	database, err := db.InitDB()
	if err != nil {
		log.Fatal(err)
	}
	defer database.Close()

	fmt.Println("DB Connected")

	queries := store.New(database)
	userService := service.NewUserService(queries)

	userService.CreateUser("John", "john@example.com")
	userService.CreateUser("Alice", "alice@example.com")

	user := userService.GetUser(1)
	fmt.Println("User:", user)

	users := userService.GetAllUsers()
	fmt.Println("Users:", users)

	userService.UpdateUser(1, "John Updated")
	userService.DeleteUser(2)
}
```

## How sqlc Flow Works

### Step-by-step

1. write SQL in `.sql` files
2. Run `sqlc generate`
3. sqlc generates Go code
4. call generated functions
5. Generated code executes SQL using `database/sql`
6. Results returned as typed structs

---

## Key Differences vs Raw SQL

| Feature        | Raw SQL | sqlc                 |
| -------------- | ------- | -------------------- |
| Query location | Go code | `.sql` files         |
| Type safety    | No      | Yes (compile-time)   |
| Boilerplate    | High    | Low                  |
| Debugging      | Manual  | Easier (SQL visible) |
| Performance    | High    | High                 |

---

# ✅ Why sqlc is Production Preferred

* SQL is explicit (no hidden queries)
* Compile-time safety
* Clean separation (SQL vs Go)
* No ORM overhead
* Works well with large teams
